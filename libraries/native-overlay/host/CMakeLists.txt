cmake_minimum_required(VERSION 3.20)
project(PraeformaOverlay LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Praeforma's Tela companion currently requires Windows")
endif()
find_package(Tela CONFIG REQUIRED)
add_executable(praeforma_overlay main.cpp review_selection.cpp review_document.cpp review_text.cpp review_panel.cpp)
target_compile_features(praeforma_overlay PRIVATE cxx_std_20)
target_compile_definitions(praeforma_overlay PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(praeforma_overlay PRIVATE Tela::Windows Tela::SceneOverlay user32)
install(TARGETS praeforma_overlay RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(praeforma_review_tests review_tests.cpp review_selection.cpp review_document.cpp review_text.cpp review_panel.cpp)
  target_compile_features(praeforma_review_tests PRIVATE cxx_std_20)
  target_link_libraries(praeforma_review_tests PRIVATE Tela::SceneOverlay)
  add_test(NAME praeforma_review COMMAND praeforma_review_tests)
endif()

if(MSVC)
  target_compile_options(praeforma_overlay PRIVATE /utf-8)
  if(BUILD_TESTING)
    target_compile_options(praeforma_review_tests PRIVATE /utf-8)
  endif()
endif()
